//! The zoo: every animal species of a seed's home planet in a line on flat ground, each walking its
//! own circle, switching walk, trot and gallop every few seconds, under a free orbit camera.
//!
//! Run: `cargo run --features dev --example zoo -- [seed] [--shots <dir>] [--find <count>]`.
//! - `seed`: the universe seed, default `DEFAULT_SEED`.
//! - `--shots <dir>`: saves `walk.png`, `trot.png` and `gallop.png` into `dir`, then exits.
//! - `--find <count>`: prints the life of the home planet of seeds 0 to `count`, then exits.
//!
//! Drag with the left mouse button to orbit, wheel to zoom.

use std::path::PathBuf;
use std::time::Duration;

use abstractions::fauna::mesh::part_mesh;
use abstractions::fauna::motion::{Detail, Drive, Motion};
use abstractions::life::body::{Body, Locomotion};
use abstractions::life::{self, Guild, Life, Species};
use abstractions::universe::Universe;
use bevy::input::mouse::{AccumulatedMouseMotion, AccumulatedMouseScroll, MouseScrollUnit};
use bevy::light::GlobalAmbientLight;
use bevy::prelude::*;
use bevy::render::view::screenshot::{Screenshot, save_to_disk};
use bevy::time::TimeUpdateStrategy;

/// A seed whose home planet holds skeletal life with walkers, found with `--find`.
const DEFAULT_SEED: u64 = 1;

/// Seconds each gait lasts before the next.
const GAIT_SECONDS: f32 = 4.0;

/// The gaits in turn: name, Froude number for walkers, body lengths a second for the others.
const GAITS: [(&str, f32, f32); 3] = [
    ("walk", 0.25, 0.4),
    ("trot", 1.4, 1.0),
    ("gallop", 3.6, 2.0),
];

/// Animals in one row.
const ROW: usize = 8;

/// Fixed step of the shots mode, seconds.
const STEP: f32 = 1.0 / 30.0;

/// One animal of the zoo.
struct Beast {
    /// Its body.
    body: Body,
    /// Its motion.
    motion: Motion,
    /// One entity per part, `None` for a part with no mesh.
    parts: Vec<Option<Entity>>,
    /// Centre of its circle.
    centre: Vec3,
    /// Radius of its circle.
    radius: f32,
}

/// The zoo and its clock.
#[derive(Resource)]
struct Zoo {
    /// Every animal.
    beasts: Vec<Beast>,
    /// Gravity of the home planet.
    gravity: f32,
    /// Seconds since the start.
    time: f32,
    /// Folder for the shots, in shots mode.
    shots: Option<PathBuf>,
    /// Frames since the last shot was asked, once all are asked.
    done: u32,
    /// Seed and life summary shown on screen.
    title: String,
}

/// The orbit camera: yaw, pitch, distance and the point it looks at.
#[derive(Component)]
struct Orbit {
    /// Yaw in radians.
    yaw: f32,
    /// Pitch in radians above the horizon.
    pitch: f32,
    /// Distance from the target in metres.
    distance: f32,
    /// Point looked at.
    target: Vec3,
}

/// Marks the text that names the gait.
#[derive(Component)]
struct Label;

/// What the command line asks.
struct Args {
    /// Universe seed.
    seed: u64,
    /// Folder for the shots.
    shots: Option<PathBuf>,
    /// Number of seeds to scan.
    find: Option<u64>,
}

/// Reads the command line.
fn args() -> Args {
    let mut out = Args {
        seed: DEFAULT_SEED,
        shots: None,
        find: None,
    };
    let mut it = std::env::args().skip(1);
    while let Some(arg) = it.next() {
        match arg.as_str() {
            "--shots" => out.shots = it.next().map(PathBuf::from),
            "--find" => out.find = it.next().and_then(|n| n.parse().ok()),
            other => out.seed = other.parse().unwrap_or(DEFAULT_SEED),
        }
    }
    out
}

/// Returns the life of the home planet of `seed` and its gravity.
fn home_life(seed: u64) -> (Option<Life>, f32) {
    let universe = Universe::new(seed);
    let id = universe.home();
    let planet = universe.planet(id);
    let star = universe.system(id.star).star;
    let gravity = life::World::of(&planet, &star).gravity;
    (life::of(&planet, &star), gravity)
}

/// Returns whether `s` is an animal with a body to draw.
fn animal(s: &Species) -> bool {
    s.role.guild != Guild::Producer && !s.body.parts.is_empty()
}

/// Prints the home life of seeds 0 to `count`.
fn find(count: u64) {
    for seed in 0..count {
        let (life, gravity) = home_life(seed);
        let Some(life) = life else {
            println!("seed {seed}: dead, g {gravity:.2}");
            continue;
        };
        let animals: Vec<&Species> = life.species.iter().filter(|s| animal(s)).collect();
        let walkers = animals
            .iter()
            .filter(|s| s.body.locomotion == Locomotion::Walk)
            .count();
        println!(
            "seed {seed}: {:?}, {} animals, {walkers} walkers, g {gravity:.2}",
            life.rung,
            animals.len()
        );
    }
}

/// Returns the radius of the circle `body` walks.
fn circle(body: &Body) -> f32 {
    (1.5 * body.length).max(1.0)
}

/// Returns the zoo of `seed`, with no part entities yet: its animals from small to large in rows
/// of `ROW`, the line of them wrapped so the camera can see them all.
fn zoo(seed: u64, shots: Option<PathBuf>) -> (Zoo, Vec<Species>) {
    let (life, gravity) = home_life(seed);
    let title = match &life {
        Some(l) => format!("seed {seed}, {:?} life, g {gravity:.2}", l.rung),
        None => format!("seed {seed}, dead planet"),
    };
    let mut species: Vec<Species> = life
        .map(|l| l.species.into_iter().filter(animal).collect())
        .unwrap_or_default();
    species.sort_by(|a, b| a.body.length.total_cmp(&b.body.length));
    let mut beasts = Vec::new();
    let mut z = 0.0;
    for (row, kinds) in species.chunks(ROW).enumerate() {
        let gap = kinds.iter().map(|s| circle(&s.body)).fold(1.0, f32::max) + 0.5;
        z -= gap;
        let mut x = 0.0;
        for (i, s) in kinds.iter().enumerate() {
            let radius = circle(&s.body);
            x += radius + 0.5;
            let centre = Vec3::new(x, 0.0, z);
            x += radius + 0.5;
            let lift = if s.body.locomotion == Locomotion::Walk {
                0.0
            } else {
                1.0
            };
            let start = centre + Vec3::new(radius, lift, 0.0);
            let seed = (row * ROW + i) as u64 + 1;
            beasts.push(Beast {
                body: s.body.clone(),
                motion: Motion::new(&s.body, start, Vec3::NEG_Z, seed),
                parts: Vec::new(),
                centre,
                radius,
            });
        }
        z -= gap;
    }
    let zoo = Zoo {
        beasts,
        gravity,
        time: 0.0,
        shots,
        done: 0,
        title,
    };
    (zoo, species)
}

/// Spawns the ground, the lights, the camera, the label and one entity per drawn part.
fn setup(
    mut commands: Commands,
    mut meshes: ResMut<Assets<Mesh>>,
    mut materials: ResMut<Assets<StandardMaterial>>,
    mut zoo: ResMut<Zoo>,
    species: Res<Kinds>,
) {
    let skin = materials.add(StandardMaterial {
        perceptual_roughness: 0.8,
        ..default()
    });
    let far = |f: fn(&Beast) -> f32| zoo.beasts.iter().map(f).fold(1.0, f32::max);
    let (span, depth) = (
        far(|b| b.centre.x + b.radius),
        far(|b| b.radius - b.centre.z),
    );
    let size = span.max(depth) + 200.0;
    let centre = Vec3::new(0.5 * span, 0.0, -0.5 * depth);
    commands.spawn((
        Mesh3d(meshes.add(Plane3d::default().mesh().size(size, size))),
        MeshMaterial3d(materials.add(Color::srgb(0.35, 0.4, 0.33))),
        Transform::from_translation(centre),
    ));
    commands.spawn((
        DirectionalLight {
            illuminance: 9000.0,
            shadow_maps_enabled: true,
            ..default()
        },
        Transform::from_xyz(-4.0, 10.0, 6.0).looking_at(Vec3::ZERO, Vec3::Y),
    ));
    commands.insert_resource(GlobalAmbientLight {
        brightness: 400.0,
        ..default()
    });
    let orbit = Orbit {
        yaw: 0.0,
        pitch: 0.6,
        distance: (0.6 * span.max(depth)).max(6.0),
        target: centre + Vec3::Y * 0.5,
    };
    commands.spawn((Camera3d::default(), orbit, Transform::default()));
    commands.spawn((
        Text::new(zoo.title.clone()),
        Label,
        Node {
            position_type: PositionType::Absolute,
            top: px(12.0),
            left: px(12.0),
            ..default()
        },
    ));
    for (beast, s) in zoo.beasts.iter_mut().zip(&species.0) {
        beast.parts = s
            .body
            .parts
            .iter()
            .map(|part| {
                let tissue = s.tissues.get(usize::from(part.tissue))?;
                let mesh = meshes.add(part_mesh(part, tissue));
                Some(
                    commands
                        .spawn((Mesh3d(mesh), MeshMaterial3d(skin.clone())))
                        .id(),
                )
            })
            .collect();
    }
}

/// The species of the zoo, in the order of its beasts.
#[derive(Resource)]
struct Kinds(Vec<Species>);

/// Returns the gait shown at `time` seconds: its index in `GAITS`.
fn gait(time: f32) -> usize {
    (time / GAIT_SECONDS) as usize % GAITS.len()
}

/// Returns the velocity that keeps `beast` on its circle at `speed`, turning left.
fn round(beast: &Beast, speed: f32) -> Vec3 {
    let off = (beast.motion.root.at - beast.centre).with_y(0.0);
    let r = off.length().max(1e-3);
    let out = off / r;
    let ahead = Vec3::Y.cross(out);
    let back = (beast.radius - r) / beast.radius;
    (ahead + out * back).normalize_or_zero() * speed
}

/// Steps every beast, poses its parts and names the gait.
fn walk(
    time: Res<Time>,
    mut zoo: ResMut<Zoo>,
    mut parts: Query<&mut Transform, Without<Orbit>>,
    mut label: Query<&mut Text, With<Label>>,
) {
    let dt = time.delta_secs().clamp(1e-4, 0.1);
    zoo.time += dt;
    let (name, fr, lengths) = GAITS[gait(zoo.time)];
    let gravity = zoo.gravity;
    for beast in &mut zoo.beasts {
        let speed = match beast.body.locomotion {
            Locomotion::Walk => (fr * gravity * beast.body.height.max(0.05)).sqrt(),
            _ => lengths * beast.body.length.max(0.2),
        };
        let drive = Drive {
            velocity: round(beast, speed),
            look: None,
            gravity,
            excite: 0.0,
            detail: Detail::Full,
        };
        beast.motion.step(&beast.body, &drive, &|_| 0.0, dt);
        let pose = beast.motion.pose(&beast.body);
        for (entity, (at, turn, scale)) in beast.parts.iter().zip(pose) {
            if let Some(mut t) = entity.and_then(|e| parts.get_mut(e).ok()) {
                *t = Transform {
                    translation: at,
                    rotation: turn,
                    scale,
                };
            }
        }
    }
    let count = zoo.beasts.len();
    let title = format!("{}, {count} animals: {name}", zoo.title);
    for mut text in &mut label {
        text.0.clone_from(&title);
    }
}

/// Orbits the camera with a left drag and zooms it with the wheel.
fn orbit(
    buttons: Res<ButtonInput<MouseButton>>,
    motion: Res<AccumulatedMouseMotion>,
    scroll: Res<AccumulatedMouseScroll>,
    mut cameras: Query<(&mut Orbit, &mut Transform)>,
) {
    for (mut o, mut t) in &mut cameras {
        if buttons.pressed(MouseButton::Left) {
            o.yaw -= 0.005 * motion.delta.x;
            o.pitch = (o.pitch + 0.005 * motion.delta.y).clamp(0.05, 1.5);
        }
        let wheel = match scroll.unit {
            MouseScrollUnit::Line => scroll.delta.y,
            MouseScrollUnit::Pixel => scroll.delta.y / 40.0,
        };
        o.distance = (o.distance * (1.0 - 0.1 * wheel)).clamp(1.0, 500.0);
        let way = Quat::from_euler(EulerRot::YXZ, o.yaw, -o.pitch, 0.0) * Vec3::Z;
        *t = Transform::from_translation(o.target + way * o.distance).looking_at(o.target, Vec3::Y);
    }
}

/// In shots mode, saves one screenshot late in each gait, then exits once they are written.
fn shoot(mut commands: Commands, mut zoo: ResMut<Zoo>, mut exit: MessageWriter<AppExit>) {
    let Some(dir) = zoo.shots.clone() else {
        return;
    };
    let shown = (zoo.time / GAIT_SECONDS) as usize;
    if shown >= GAITS.len() {
        zoo.done += 1;
        let written = GAITS
            .iter()
            .all(|g| dir.join(format!("{}.png", g.0)).exists());
        if (written && zoo.done > 10) || zoo.done > 300 {
            exit.write(AppExit::Success);
        }
        return;
    }
    let late = zoo.time % GAIT_SECONDS;
    if (GAIT_SECONDS - 0.5..GAIT_SECONDS - 0.5 + STEP).contains(&late) {
        let path = dir.join(format!("{}.png", GAITS[shown].0));
        commands
            .spawn(Screenshot::primary_window())
            .observe(save_to_disk(path));
    }
}

/// Opens the zoo, or scans seeds with `--find`.
fn main() -> AppExit {
    let args = args();
    if let Some(count) = args.find {
        find(count);
        return AppExit::Success;
    }
    if let Some(dir) = &args.shots
        && let Err(e) = std::fs::create_dir_all(dir)
    {
        eprintln!("cannot make {}: {e}", dir.display());
        return AppExit::error();
    }
    let shots = args.shots.is_some();
    let (zoo, species) = zoo(args.seed, args.shots);
    println!("{}: {} animals", zoo.title, zoo.beasts.len());
    let mut app = App::new();
    app.add_plugins(DefaultPlugins)
        .insert_resource(zoo)
        .insert_resource(Kinds(species))
        .insert_resource(ClearColor(Color::srgb(0.62, 0.72, 0.82)))
        .add_systems(Startup, setup)
        .add_systems(Update, (walk, orbit, shoot).chain());
    if shots {
        app.insert_resource(TimeUpdateStrategy::ManualDuration(Duration::from_secs_f32(
            STEP,
        )));
    }
    app.run()
}
